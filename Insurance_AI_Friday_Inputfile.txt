## Phase 6: Create the SQLite database

Create:

```text
backend/services/database.py
```

```python
from pathlib import Path

import pandas as pd
from sqlalchemy import create_engine


BASE_DIR = Path(__file__).resolve().parent.parent.parent
CSV_FILE = BASE_DIR / "data" / "policies.csv"
DB_FILE = BASE_DIR / "insurance.db"

engine = create_engine(f"sqlite:///{DB_FILE}")


def initialize_database():
    policies = pd.read_csv(CSV_FILE)

    policies.to_sql(
        "policies",
        engine,
        if_exists="replace",
        index=False
    )

    print(f"Loaded {len(policies)} policies into SQLite.")


def get_all_policies():
    return pd.read_sql("SELECT * FROM policies", engine)


if __name__ == "__main__":
    initialize_database()
```

## Phase 8: Build the eligibility engine

Create:

```text
backend/engines/eligibility_engine.py
```

```python
def is_eligible(customer, policy):

    if not policy["min_age"] <= customer.age <= policy["max_age"]:
        return False

    if (
        customer.dependents is not None
        and customer.dependents > 0
        and not policy["family_supported"]
    ):
        return False

    if (
        customer.dependents is not None
        and customer.dependents > policy["max_dependents"]
    ):
        return False

    if (
        customer.pre_existing_conditions is True
        and not policy["pre_existing_conditions"]
    ):
        return False

    return True


def get_eligible_policies(customer, policies):
    return [
        policy
        for _, policy in policies.iterrows()
        if is_eligible(customer, policy)
    ]
```

## Phase 9: Build recommendation engine

Create:

```text
backend/engines/recommendation_engine.py
```

```python
def calculate_score(customer, policy):
    score = 0
    reasons = []

    if policy["monthly_premium"] <= customer.monthly_budget:
        score += 30
        reasons.append("Fits within your monthly budget.")

    coverage_ratio = min(
        policy["coverage_amount"] / customer.desired_coverage,
        1
    )

    score += coverage_ratio * 30

    if policy["coverage_amount"] >= customer.desired_coverage:
        reasons.append("Meets your desired coverage amount.")

    if customer.dependents > 0 and policy["family_supported"]:
        score += 20
        reasons.append("Supports family coverage.")

    if (
        customer.pre_existing_conditions
        and policy["pre_existing_conditions"]
    ):
        score += 10
        reasons.append("Supports declared pre-existing conditions.")

    if (
        customer.international_coverage
        and policy["geographic_coverage"] == "International"
    ):
        score += 10
        reasons.append("Includes international coverage.")

    return round(score, 2), reasons


def recommend_policies(customer, eligible_policies, top_n=3):
    recommendations = []

    for policy in eligible_policies:
        score, reasons = calculate_score(customer, policy)

        recommendations.append(
            {
                "policy": dict(policy),
                "score": score,
                "reasons": reasons
            }
        )

    recommendations.sort(
        key=lambda item: item["score"],
        reverse=True
    )

    return recommendations[:top_n]
```

```python
from fastapi import FastAPI

from backend.engines.eligibility_engine import get_eligible_policies
from backend.engines.recommendation_engine import recommend_policies
from backend.models.customer import CustomerProfile
from backend.services.database import get_all_policies


app = FastAPI(
    title="Insurance Policy Recommendation API",
    version="1.0"
)


@app.get("/")
def home():
    return {
        "message": "Insurance AI Chatbot API is running"
    }


@app.post("/recommend")
def recommend(customer: CustomerProfile):

    policies = get_all_policies()

    eligible = get_eligible_policies(
        customer,
        policies
    )

    recommendations = recommend_policies(
        customer,
        eligible,
        top_n=3
    )

    return {
        "eligible_policy_count": len(eligible),
        "recommendations": recommendations
    }
```

```python
import ollama


MODEL_NAME = "llama3.2:3b"


def generate_response(prompt: str) -> str:

    response = ollama.chat(
        model=MODEL_NAME,
        messages=[
            {
                "role": "system",
                "content": """
You are an insurance policy assistant.

Explain insurance concepts clearly and simply.
Never invent policy coverage, exclusions, premiums,
eligibility criteria, or claim outcomes.
"""
            },
            {
                "role": "user",
                "content": prompt
            }
        ]
    )

    return response["message"]["content"]
```

```text

    ↓
Embeddings
    ↓
ChromaDB
    ↓
User asks question
    ↓
Semantic retrieval
    ↓
Top relevant document chunks
    ↓
Ollama receives only retrieved context
    ↓
Grounded answer
```

The embedding model can be:

```text
all-MiniLM-L6-v2
```
